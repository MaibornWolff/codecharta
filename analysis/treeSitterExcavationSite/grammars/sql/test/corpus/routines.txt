CREATE PROCEDURE dbo.GetOrders @CustomerId INT, @Status NVARCHAR(10) = NULL, @Limit INT = 10
AS
BEGIN
    SET NOCOUNT ON;
    SELECT o.Id FROM dbo.Orders o WHERE o.CustomerId = @CustomerId;
END
GO

CREATE OR REPLACE PROCEDURE raise_salary (p_emp_id IN NUMBER, p_amount IN NUMBER DEFAULT 0) IS
  v_salary NUMBER;
  CURSOR c_staff IS SELECT id FROM employees WHERE dept = 10;
  FUNCTION bonus(p_id IN NUMBER) RETURN NUMBER IS
  BEGIN
    RETURN 0;
  END bonus;
BEGIN
  UPDATE employees SET salary = salary + p_amount WHERE id = p_emp_id;
END raise_salary;

    LANGUAGE plpgsql
    AS $function$
BEGIN
  NEW.updated_at := now();
  RETURN NEW;
END;
$function$ SECURITY DEFINER;
